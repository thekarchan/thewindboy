{{- $u := urls.Parse .Destination -}} {{- if $u.IsAbs -}} {{- else -}} {{- $cleanPath := strings.TrimPrefix "./" $u.Path -}} {{- $img := or (.Page.Resources.GetMatch $u.Path) (.Page.Resources.GetMatch $cleanPath) -}} {{- if and $img (not (strings.HasSuffix $cleanPath ".svg")) (not (strings.HasSuffix $cleanPath ".gif")) -}} {{- $optimized := $img -}} {{- if gt $img.Width 1200 -}} {{- $optimized = $img.Resize "1200x webp q85" -}} {{- else if and (ne $img.MediaType.SubType "webp") (ne $img.MediaType.SubType "svg") -}} {{- $optimized = $img.Resize (printf "%dx webp q85" $img.Width) -}} {{- end -}} {{- else -}} {{- $src := .Destination -}} {{- if and (not (hasPrefix $u.Path "/")) (not (hasPrefix $u.Path "http://")) (not (hasPrefix $u.Path "https://")) -}} {{- with .Page.File -}} {{- $resolved := path.Join "/" .Dir $cleanPath -}} {{- $src = $resolved | relURL -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
